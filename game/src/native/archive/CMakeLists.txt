cmake_minimum_required(VERSION 3.22)
project(mod_archive LANGUAGES C CXX)
set(GODOTCPP_DIR "" CACHE PATH "godot-cpp checkout at the documented revision")
set(GODOTCPP_TARGET "template_release" CACHE STRING "" FORCE)
set(GODOTCPP_API_VERSION "4.7" CACHE STRING "" FORCE)
set(GODOTCPP_BUILD_PROFILE "${CMAKE_CURRENT_SOURCE_DIR}/build_profile.json" CACHE FILEPATH "" FORCE)
set(GODOTCPP_USE_HOT_RELOAD OFF CACHE BOOL "" FORCE)
add_subdirectory("${GODOTCPP_DIR}" godot-cpp)
file(GLOB SDK_SOURCES CONFIGURE_DEPENDS vendor/lzma-26.04/*.c)
add_library(mod_archive SHARED mod_archive.cpp ${SDK_SOURCES})
target_include_directories(mod_archive PRIVATE vendor/lzma-26.04)
target_compile_definitions(mod_archive PRIVATE Z7_EXTRACT_ONLY)
target_link_libraries(mod_archive PRIVATE godot-cpp)
target_compile_features(mod_archive PRIVATE cxx_std_17)
set_target_properties(mod_archive PROPERTIES PREFIX "lib" C_VISIBILITY_PRESET hidden CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(mod_archive PRIVATE "-ffile-prefix-map=${CMAKE_CURRENT_SOURCE_DIR}=src/native/archive" "-ffile-prefix-map=${GODOTCPP_DIR}=godot-cpp" "-ffile-prefix-map=${CMAKE_BINARY_DIR}=build/mod-archive")
endif()
if(MINGW)
  target_link_options(mod_archive PRIVATE -static-libgcc -static-libstdc++ -static -Wl,--no-insert-timestamp)
endif()
