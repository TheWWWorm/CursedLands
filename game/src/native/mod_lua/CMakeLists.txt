cmake_minimum_required(VERSION 3.22)
project(mod_lua LANGUAGES C CXX)
set(GODOTCPP_DIR "" CACHE PATH "godot-cpp checkout (same revision as terrain_search)")
set(GODOTCPP_TARGET "template_release" CACHE STRING "" FORCE)
set(GODOTCPP_API_VERSION "4.7" CACHE STRING "" FORCE)
set(GODOTCPP_BUILD_PROFILE "${CMAKE_CURRENT_SOURCE_DIR}/build_profile.json" CACHE FILEPATH "" FORCE)
set(GODOTCPP_USE_HOT_RELOAD OFF CACHE BOOL "" FORCE)
add_subdirectory("${GODOTCPP_DIR}" godot-cpp)
file(GLOB LUA_SOURCES CONFIGURE_DEPENDS vendor/lua-5.1.5/*.c)
list(FILTER LUA_SOURCES EXCLUDE REGEX "/(lua|luac|print|linit|loadlib|liolib|loslib|ldblib)\\.c$")
add_library(mod_lua SHARED mod_lua.cpp ${LUA_SOURCES})
target_include_directories(mod_lua PRIVATE vendor/lua-5.1.5)
target_link_libraries(mod_lua PRIVATE godot-cpp)
target_compile_features(mod_lua PRIVATE cxx_std_17)
set_target_properties(mod_lua PROPERTIES PREFIX "lib" C_VISIBILITY_PRESET hidden CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(mod_lua PRIVATE -ffp-contract=off "-ffile-prefix-map=${CMAKE_CURRENT_SOURCE_DIR}=src/native/mod_lua" "-ffile-prefix-map=${GODOTCPP_DIR}=godot-cpp" "-ffile-prefix-map=${CMAKE_BINARY_DIR}=build/mod-lua")
endif()
if(MINGW)
  target_link_options(mod_lua PRIVATE -static-libgcc -static-libstdc++ -static -Wl,--no-insert-timestamp)
endif()
